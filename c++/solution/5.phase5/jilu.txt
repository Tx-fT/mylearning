#include<bits/stdc++.h>
using namespace std;
const int MOD=1e9+7;

// fast pow
long long qpow(long long a,long long b){
    long long r=1;
    while(b){
        if(b&1) r=r*a%MOD;
        a=a*a%MOD; b>>=1;
    }
    return r;
}

void solve(){
    int n,d;
    cin>>n>>d;

    // d=1: any subset works, answer = 2^(2n)
    if(d==1){
        cout<<qpow(2,2ll*n)<<'\n';
        return;
    }

    // number of states: 2d-1
    // state encoding:
    //   0: empty (no black cell in last d-1 cols)
    //   1..d-1: row-0 black at position (s-1), meaning column cur-(s-1)-1 = cur-s
    //   d..2d-2: row-1 black at position (s-d)
    int S=2*d-1;
    vector<long long> dp(S,0);
    dp[0]=1;  // before first column: empty window

    int fullMask=(1<<(d-1))-1;      // mask for d-1 bits
    int crossMask=(1<<max(0,d-2))-1; // mask for cross-row check (d-2 bits)

    for(int col=0;col<n;col++){
        vector<long long> ndp(S,0);
        for(int s=0;s<S;s++){
            long long cur=dp[s];
            if(!cur) continue;

            // decode state into two bitmasks
            int m0=0,m1=0;
            if(s!=0){
                if(s<d) m0=1<<(s-1);
                else    m1=1<<(s-d);
            }

            // try 3 column configs: 0=empty, 1=row0 black, 2=row1 black
            for(int c=0;c<3;c++){
                // ---- validity checks ----
                if(c==1){  // place black in row 0
                    if(m0) continue;                    // same-row too close
                    if(m1 & crossMask) continue;        // cross-row too close
                }
                if(c==2){  // place black in row 1
                    if(m1) continue;
                    if(m0 & crossMask) continue;
                }

                // ---- compute new masks ----
                int nm0=((m0<<1)|(c==1)) & fullMask;
                int nm1=((m1<<1)|(c==2)) & fullMask;//给下一波的状态

                // ---- encode new state ----
                int ns;
                if(nm0==0 && nm1==0) ns=0;
                else if(nm1==0)       ns=__builtin_ctz(nm0)+1;//最近的黑格
                else                  ns=__builtin_ctz(nm1)+d;

                ndp[ns]=(ndp[ns]+cur)%MOD;//在最近雷基础上，加上符合条件的状态
            }
        }
        dp.swap(ndp);
    }

    long long ans=0;
    for(long long x:dp) ans=(ans+x)%MOD;
    cout<<ans<<'\n';
}

    ios::sync_with_stdio(0);
    cin.tie(0);cout.tie(0);
    int T;cin>>T;
    while(T--) solve();
}

struct Interval {
    int l, r, v;
};

void solve() {
    int n, m;
    cin >> n >> m;

    vector<Interval> cons(m);
    for (int i = 0; i < m; i++) {
        cin >> cons[i].l >> cons[i].r >> cons[i].v;
    }

    const int INF = 2e9;
    vector<int> a(n + 1, INF);

    // DSU: nxt[i] = next unset position (>= i)
    vector<int> nxt(n + 2);
    iota(nxt.begin(), nxt.end(), 0);//全部指向自己

    function<int(int)> find = [&](int x) -> int {
        return nxt[x] == x ? x : nxt[x] = find(nxt[x]);
    };//函数

    // Sort by v ascending: each position gets the MINIMUM v among its intervals
    sort(cons.begin(), cons.end(),
         [](const Interval& x, const Interval& y) { return x.v < y.v; });

    for (auto [l, r, v] : cons) {
        for (int i = find(l); i <= r; i = find(i)) {//并查集标记用法
            a[i] = v;
            nxt[i] = i + 1;
        }
    }

    // Positions not covered by any constraint: set to 1
    for (int i = 1; i <= n; i++) {
        if (a[i] == INF) a[i] = 1;
    }

    // Group positions by value for verification
    map<int, vector<int>> pos;
    for (int i = 1; i <= n; i++) {
        pos[a[i]].push_back(i);
    }

    // Verify: each constraint must have at least one position == v in [l, r]
    bool ok = true;
    for (auto [l, r, v] : cons) {
        auto it = pos.find(v);
        if (it == pos.end()) { ok = false; break; }
        auto& vec = it->second;
        auto lb = lower_bound(vec.begin(), vec.end(), l);//二分厉害
        if (lb == vec.end() || *lb > r) { ok = false; break; }
    }

    if (!ok) {
        cout << "No\n";
    } else {
        cout << "Yes\n";
        for (int i = 1; i <= n; i++) {
            cout << a[i] << " \n"[i == n];
        }
    }
}

    return 0;

void solve(){
    int n, k;
    cin >> n >> k;
    string a, b;
    cin >> a >> b;

    ull total = (1ULL << k) + 1;
    ull cnt3;
    if(k & 1){
        cnt3 = ((1ULL << k) + 1) / 3;   // (-1)^k = -1
    } else {
        cnt3 = ((1ULL << k) - 1) / 3;
    }
    ull cnt1 = (total - cnt3) / 2;
    ull cnt2 = cnt1;

    ull b1 = count(a.begin(), a.end(), '1');
    ull b2 = count(b.begin(), b.end(), '1');
    ull b3 = 0;
    for(int i = 0; i < n; i++){
        if(a[i] != b[i]) b3++;
    }

    ull ans = 0;
    ans += cnt1 * b1 * (n - b1);
    ans += cnt2 * b2 * (n - b2);
    ans += cnt3 * b3 * (n - b3);

    cout << ans << '\n';
}

void dfs1(int now,int f){
    dp[now]=1;
    for(int i:g[now]){
        if(i==f){
            continue;
        }
        dfs1(i,now);
        dp[now]+=dp[i];
    }

        ans[1]++;
    }

void dfs2(int now,int f){
    for(int i:g[now]){
        if(i==f){
            continue;
        }
        ans[i]=ans[now];
        int tep=dp[i];
        dp[now]-=dp[i];
        dp[i]=n;
        if(tep<k){
            ans[i]+=1;
        }
        if(dp[now]<k){
            ans[i]-=1;
        }
        dfs2(i,now);
        dp[now]=n;
        dp[i]=tep;
    }
}

        int cnt = len - 2 * x;
        if(i == 1 || i == n + 1) cnt += x;

        if(sum >= k) return true;
    }

}

#include <bits/stdc++.h>
using namespace std;

const int MOD = 998244353;

long long modpow(long long a, long long b) {
    long long res = 1;
    while (b) {
        if (b & 1) res = res * a % MOD;
        a = a * a % MOD;
        b >>= 1;
    }
    return res;
}

long long modinv(long long x) {
    return modpow(x, MOD - 2);
}

void solve() {
    int n;
    cin >> n;
    vector<long long> a(n), b(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];

    if (n == 1) {
        cout << "0\n";
        return;
    }

    // 生成所有 n^2 个比值 (b_q, b_p) 即 b_q / b_p
    vector<pair<long long, long long>> ratios;
    ratios.reserve((size_t)n * n);
    for (int q = 0; q < n; q++)
        for (int p = 0; p < n; p++)
            ratios.emplace_back(b[q], b[p]);

    // 按分数大小排序: b_q1/b_p1 < b_q2/b_p2
    sort(ratios.begin(), ratios.end(), [](const auto& x, const auto& y) {
        return x.first * y.second < y.first * x.second;
    });

    // 生成所有查询 (a_i, a_j) for i<j
    struct Query {
        long long ai, aj;
        bool ai_gt_aj;
    };
    vector<Query> queries;
    queries.reserve((size_t)n * (n - 1) / 2);
    for (int i = 0; i < n; i++)
        for (int j = i + 1; j < n; j++)
            queries.push_back({a[i], a[j], a[i] > a[j]});

    // 按分数大小排序: a_i1/a_j1 < a_i2/a_j2
    sort(queries.begin(), queries.end(), [](const Query& x, const Query& y) {
        return x.ai * y.aj < y.ai * x.aj;
    });

    long long total = 0;
    int ptr = 0;
    int m = n * n;

    for (const auto& q : queries) {
        // 推进 ptr，跳过所有 b_q/b_p < a_i/a_j 的比值
        while (ptr < m) {
            auto& r = ratios[ptr];
            // r.first/r.second < q.ai/q.aj  <=>  r.first * q.aj < q.ai * r.second
            if (r.first * q.aj < q.ai * r.second)
                ptr++;
            else
                break;
        }
//小/大--小/大--双指针
        long long cnt = ptr;
        if (q.ai_gt_aj) cnt -= n; // 排除 p=q（比值=1）的 n 个对角线
        total = (total + cnt) % MOD;
    }

    long long denom = (long long)n * (n - 1) % MOD;//分母
    long long ans = total * modinv(denom) % MOD;
    cout << ans << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T;
    cin >> T;
    while (T--) solve();
    return 0;
}

		int u,v;

ll C(ll m,ll n){
	if(n>m||n<0) return 0;
	if(n>m-n) n=m-n;
	ll res=1;
	for(ll i=1;i<=n;i++) res=res*(m-i+1)/i;
	return res;
}

		d--;
		ans++;
	}
	ll i=d;
	for(;;i--){
		ll left=k-i;
		if(left>=i-1) break;
		for(ll j=0;j<=left&&j<i;j++) ans+=C(i-1,j);
	}
	for(;i>=1;i--) ans+=(1LL<<(i-1));
	cout<<n-ans<<"\n";
}

    int res=1;
    while(x){
        if(x&1){
            res=(1ll*b*res)%mod;//注意开ll超栈
        }
        b=(1ll*b*b)%mod;

    }

}

#include<bits/stdc++.h>
using namespace std;
using ll=long long;

const ll MOD=998244353;  // 质数，可用费马小定理

// 快速幂：计算 2^x mod MOD
// x 可能高达 1e18，利用费马小定理：2^{MOD-1} ≡ 1 (mod MOD)
// 所以 2^x ≡ 2^{x % (MOD-1)} (mod MOD)
ll pow2(ll x){
    x%=(MOD-1);
    ll res=1,base=2;
    while(x){
        if(x&1) res=res*base%MOD;
        base=base*base%MOD;
        x>>=1;
    }
    return res;
}

// 缓存 pow2 的结果，避免重复计算
map<ll,ll> memo_pow2;
ll get_pow2(ll x){
    if(memo_pow2.count(x)) return memo_pow2[x];
    return memo_pow2[x]=pow2(x);
}

/*
 * 核心公式推导：
 *
 * 对于线段树的每个节点 v，设其子树包含 L 个叶子
 * 以 v 为 LCA 的子集数量 = 2^L - 2^{L_left} - 2^{L_right} + 1
 * （所有子集 - 全在左子树的 - 全在右子树的 + 空集被多减了一次补回来）
 * 节点 v 对答案的贡献 = v * count(L)
 *
 * 对于子树，定义：
 * A(len) = 以 label=1 为根、有 len 个叶子的子树的 总贡献
 * B(len) = 该子树中所有节点的 2^{depth} * count(L_node) 之和
 *
 * 关键观察：对于根 label=v 的子树，节点 label = 标准label + (v-1)*2^{depth}
 * 所以 F(len, v) = A(len) + (v-1) * B(len)
 *
 * 递归：
 * left_len = (len+1)/2 (ceil), right_len = len/2 (floor)
 * count(len) = 2^len - 2^{left_len} - 2^{right_len} + 1  (len>1)
 * count(1)   = 1  (叶子节点)
 *
 * A(len) = count(len) + A(left_len) + A(right_len) + B(left_len) + 2*B(right_len)
 * （左子树 root=v*2,   v=2 → v-1=1，偏移系数为 1）
 * （右子树 root=v*2+1, v=3 → v-1=2，偏移系数为 2）
 *
 * B(len) = count(len) + 2*B(left_len) + 2*B(right_len)
 * （子节点深度+1，所以 2^{depth} 乘以 2）
 *
 * 最终答案 = A(n)，因为根节点 label=1，偏移系数为 0
 */

map<ll,pair<ll,ll>> memo;  // len -> {A(len), B(len)}

pair<ll,ll> solve(ll n){
    if(n==1) return {1, 1};  // A(1)=1(叶子自身贡献), B(1)=2^0*1=1

    if(memo.count(n)) return memo[n];

    ll L=(n+1)/2;  // 左子树叶子数 = ceil(n/2)
    ll R=n/2;      // 右子树叶子数 = floor(n/2)

    auto [aL, bL]=solve(L);
    auto [aR, bR]=solve(R);

    // count(n) = 2^n - 2^L - 2^R + 1
    ll cnt=(get_pow2(n) - get_pow2(L) - get_pow2(R) + 1) % MOD;
    cnt=(cnt+MOD) % MOD;  // 防负数

    // A(n) = cnt + A(L) + A(R) + 1*B(L) + 2*B(R)
    ll A = (cnt + aL + aR + bL + 2*bR) % MOD;

    // B(n) = cnt + 2*B(L) + 2*B(R)
    ll B = (cnt + 2*bL + 2*bR) % MOD;

    return memo[n] = {A, B};
}

                }

typedef long long ll;
struct node{
    int idx,cnt=0;
};
void solve(){
    int n;
    cin>>n;
    vector<int> q;
    for(int i=1;i<=n;i++){
        int t;
        cin>>t;
        q.push_back(t);
    }
    sort(q.begin(),q.end());
    map<int,int> m;
    vector<node> a(n+1);
    int cnt=0;
    for(int i:q){
        if(m[i]==0){
            m[i]=++cnt;
            a[cnt].idx=i;
        }
        a[m[i]].cnt++;
    }
    int pre=0;
    ll ans=0;
    a[0].idx=-1;
    a[0].cnt=0;
    for(int i=1;i<=cnt;i++){
        if(a[i].idx>a[i-1].idx+1){
            pre=0;
        }
        if(a[i].cnt>pre){
            ans+=(a[i].cnt-pre);
        }
        pre=a[i].cnt;
    }
    cout<<ans<<endl;
}
signed main(){
    ios::sync_with_stdio(false);
    cin.tie(0);   
    int t;
    cin>>t;
    while(t--){
        solve();
    }
    return 0;
}

vector<int> g[200010];
int d[200010];
int mi=0x3f3f3f3f;
void dfs(int n,int p,int dep){
    if(dep>mi){
        return;
    }
    for(int i:g[n]){
        if(i==p) continue;
        if(d[i]>dep){
            d[i]=dep;
            dfs(i,n,dep+1);
        }
    }
}

        d[i]=0x3f3f3f3f;
    }
    g[1].clear();
    d[1]=0x3f3f3f3f;
    d[k]=0;
    mi=0x3f3f3f3f;
    for(int i=1;i<n;i++){
        int a,b;
        cin>>a>>b;
        g[a].push_back(b);
        g[b].push_back(a);
    }
    for(int i=1;i<=n;i++){
        if(i!=1){
            mi=min(mi,d[v[i]]);
            cout<<mi<<" ";
        }
        d[v[i]]=0;
        dfs(v[i],-1,1);
    }
    cout<<endl;
}

int lowbit(int x){
    return x&-x;
}
ll a[200010];
ll tr[200010];
void add(ll x,int p){
    for(int i=p;i<mx;i+=lowbit(i)){
        tr[i]+=x;
    }
}
ll query(int l,int r){
    ll left=0;
    for(int i=l-1;i>0;i-=lowbit(i)){
        left+=tr[i];
    }
    ll right=0;
    for(int i=r;i>0;i-=lowbit(i)){
        right+=tr[i];
    }
    return right-left;
}

ll qpow(ll x,int b){
    ll res=1;
    while(x){
        if(x&1){
            res=(1ll*res*b)%mod;
        }
        b=(1ll*b*b)%mod;
        x>>=1;
    }
    return res;
}
ll inv(ll b){
    return qpow(mod-2,b);
}
void solve(){
    cin>>n>>m;
    p[0]=1;
    p_[0]=1;
    for(int i=1;i<=n;i++){
        p[i]=(p[i-1]*i)%mod;
        p_[i]=inv(p[i]);
    }
    for(int i=1;i<=n;i++){
        cin>>a[i];
        // a[i]+=1e9;
    }
    if(m==1){

        return;
    }
    sort(a+1,a+n+1);
    ll ans=0;
    a[0]=0;
    for(int i=1;i<=n;i++){
        a[i]+=a[i-1];
    }
    for(int i=n;i>=m;i--){
        ll c=a[n]-a[i-1]-a[n-i+1];
        ans = (ans + (c % mod) * p[i-2] % mod * p_[m-2] % mod * p_[i-m]) % mod;
    }
    cout<<ans<<endl;
}

        else cout<<"No";

    }

    cout<<endl;
}
